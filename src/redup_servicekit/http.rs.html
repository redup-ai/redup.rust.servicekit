35/// Build the axum router serving `/metrics`, `/livez`, `/readyz`.
36/// The readiness probe must be configured via [`ServiceKit::set_ready_probe`].
37pub fn router(kit: Arc<ServiceKit>) -> Router {
38    Router::new()
39        .route("/metrics", get(metrics_handler))
40        .route("/livez", get(livez_handler))
41        .route("/readyz", get(readyz_handler))
42        .with_state(HttpState::new(kit))
43}
44
45async fn metrics_handler(State(state): State<HttpState>) -> impl IntoResponse {
46    let body = state.kit.metrics_text();
47    (
48        StatusCode::OK,
49        [("content-type", "text/plain; version=0.0.4")],
50        String::from_utf8_lossy(&body).to_string(),
51    )
52}
53
54async fn livez_handler() -> impl IntoResponse {
55    (StatusCode::OK, "OK")
56}
57
58async fn readyz_handler(State(state): State<HttpState>) -> impl IntoResponse {
59    if state.kit.is_ready() {
60        (StatusCode::OK, "OK")
61    } else {
62        (StatusCode::SERVICE_UNAVAILABLE, "NOT_READY")
63    }
64}

66/// Tonic-gRPC compatible metrics layer for plain HTTP services (axum/axum).
67///
68/// Analog of [`crate::grpc::GrpcMetricsLayer`] for REST handlers
69/// (`redup.python.servicekit` `request_guard`): per request it increments
70/// `request___method__<m>`, registers a task, measures duration and
71/// request/response sizes, and on response increments
72/// `processed___method__<m>___status__success|failure`, on failure also
73/// `errors___method__<m>___type__other`, observes `time full`,
74/// `request_size`, `response_size` (EMA + `stats_hist_*`).
75///
76/// Method name is the last path segment (`/api/v1/billing/summary` ->
77/// `summary`); the full route is available via
78/// [`http_method_name`]. Sizes come from `size_hint().exact()`.
79#[derive(Clone)]
80pub struct HttpMetricsLayer {
81    kit: Arc<ServiceKit>,
82}
83
84impl HttpMetricsLayer {
85    pub fn new(kit: Arc<ServiceKit>) -> Self {
86        Self { kit }
87    }
88}
89
90impl<S> Layer<S> for HttpMetricsLayer {
91    type Service = HttpMetricsService<S>;
92
93    fn layer(&self, inner: S) -> Self::Service {
94        HttpMetricsService {
95            inner,
96            kit: self.kit.clone(),
97        }
98    }
99}

101/// RAII guard for the in-flight HTTP task metric.
102///
103/// `HttpMetricsService::call` must finish the tracked task exactly once per
104/// request. A regular completion calls [`finish`](Self::finish) (records the
105/// duration); if the handler future is dropped mid-flight — client gone,
106/// proxy reset, timeout at the edge — the guard's `Drop` discards the task
107/// so `tasks_count` / `current_tasks_max_time_spend` cannot grow forever.
108struct TaskGuard {
109    kit: Arc<ServiceKit>,
110    id: Option<String>,
111}
112
113impl TaskGuard {
114    fn start(kit: &Arc<ServiceKit>) -> Self {
115        let id = kit.task_tracker().start();
116        Self {
117            kit: kit.clone(),
118            id: Some(id),
119        }
120    }
121
122    fn finish(&mut self) {
123        if let Some(id) = self.id.take() {
124            self.kit.task_tracker().finish(&id);
125        }
126    }
127}
128
129impl Drop for TaskGuard {
130    fn drop(&mut self) {
131        if let Some(id) = self.id.take() {
132            self.kit.task_tracker().discard(&id);
133        }
134    }
135}

137/// Service wrapper produced by [`HttpMetricsLayer`].
138#[derive(Clone)]
139pub struct HttpMetricsService<S> {
140    inner: S,
141    kit: Arc<ServiceKit>,
142}
143
144impl<S, B> Service<Request<B>> for HttpMetricsService<S>
145where
146    S: Service<Request<B>, Response = Response<B>> + Clone + Send + 'static,
147    S::Future: Send + 'static,
148    B: HttpBody + Send + 'static,
149{
150    type Response = Response<B>;
151    type Error = S::Error;
152    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
153
154    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
155        self.inner.poll_ready(cx)
156    }
157
158    fn call(&mut self, req: Request<B>) -> Self::Future {
159        // Prefer the matched route template (bounded cardinality: one series
160        // per registered route), so dynamic segments like `/tasks/{uuid}`
161        // never become their own metric series. Fall back to the URI-path
162        // heuristic only for requests that did not match a route (404/probe).
163        let method = req
164            .extensions()
165            .get::<axum::extract::MatchedPath>()
166            .map(|mp| mp.as_str().to_string())
167            .unwrap_or_else(|| http_method_name(req.uri().path()));
168        let kit = self.kit.clone();
169        let request_size = req.body().size_hint().exact().unwrap_or(0) as f64;
170        let mut inner = self.inner.clone();
171
172        let start = std::time::Instant::now();
173        kit.init_method(&method);
174        kit.inc_stat(&format!("request___method__{method}"), 1.0);
175        let mut guard = TaskGuard::start(&kit);
176
177        let future = async move {
178            let result = inner.call(req).await;
179            let elapsed = start.elapsed().as_secs_f64();
180            guard.finish();
181
182            match result {
183                Ok(response) => {
184                    let failed = response.status().as_u16() >= 400;
185                    let response_size = response.body().size_hint().exact().unwrap_or(0) as f64;
186                    let status = if failed { "failure" } else { "success" };
187                    kit.inc_stat(
188                        &format!("processed___method__{method}___status__{status}"),
189                        1.0,
190                    );
191                    if failed {
192                        kit.inc_stat(&format!("errors___method__{method}___type__other"), 1.0);
193                    }
194                    kit.observe_stat(&format!("time full___method__{method}"), elapsed);
195                    kit.observe_stat(&format!("request_size___method__{method}"), request_size);
196                    kit.observe_stat(&format!("response_size___method__{method}"), response_size);
197                    kit.set_task_status(
198                        if failed {
199                            crate::monitor::TaskStatus::OtherException
200                        } else {
201                            crate::monitor::TaskStatus::Ok
202                        },
203                        &method,
204                    );
205                    Ok(response)
206                }
207                Err(error) => {
208                    kit.inc_stat(
209                        &format!("processed___method__{method}___status__failure"),
210                        1.0,
211                    );
212                    kit.inc_stat(&format!("errors___method__{method}___type__other"), 1.0);
213                    kit.observe_stat(&format!("time full___method__{method}"), elapsed);
214                    kit.observe_stat(&format!("request_size___method__{method}"), request_size);
215                    kit.set_task_status(crate::monitor::TaskStatus::OtherException, &method);
216                    Err(error)
217                }
218            }
219        };
220        Box::pin(future)
221    }
222}

224/// Derive a metric method name from a raw URI path.
225///
226/// This is only a **fallback** for requests that did not match a registered
227/// route, so `MatchedPath` is absent (e.g. a 404 / unknown-path probe).
228/// The canonical source is [`MatchedPath`] (the registered route template),
229/// which has bounded cardinality. Here we keep the last path segment but
230/// collapse clearly dynamic segments (uuid-like, long hex, bare number) to
231/// `id` so an unknown dynamic path still cannot explode the metric series.
232pub fn http_method_name(path: &str) -> String {
233    let trimmed = path.trim_end_matches('/');
234    let last = trimmed.rsplit('/').find(|s| !s.is_empty()).unwrap_or("root");
235    if is_dynamic_segment(last) {
236        "id".to_string()
237    } else {
238        last.to_string()
239    }
240}
241
242/// Heuristic only for the fallback path: collapse a segment that is clearly
243/// dynamic (a generated id) to a single bounded label `id`.
244fn is_dynamic_segment(seg: &str) -> bool {
245    if seg.is_empty() {
246        return true;
247    }
248    // Bare integer.
249    if seg.chars().all(|c| c.is_ascii_digit()) {
250        return true;
251    }
252    let hex_count = seg.chars().filter(|c| c.is_ascii_hexdigit()).count();
253    // Hex-id: entirely hex chars, optionally separated by '-'. This covers
254    // short uuid-like ids ("deadbeef-1234", "abcd1234") as well as long
255    // hashes, while leaving static segments like "summary"/"billing" (which
256    // contain non-hex letters) untouched. Only the last path segment reaches
257    // here, so an aggressive match cannot misfire on the api/v1 prefixes.
258    let all_hex = seg.chars().all(|c| c.is_ascii_hexdigit() || c == '-');
259    // UUID shape (dashes optional) or a long hex token.
260    let uuidish = (seg.len() >= 32 && hex_count >= 32) || (seg.contains('-') && seg.len() >= 36);
261    // Long hex token without separators (hash ids).
262    let long_hex = seg.len() >= 24 && hex_count == seg.len();
263    (all_hex && hex_count >= 8) || uuidish || long_hex
264}

276    /// Parse a scalar Prometheus metric (the first `{name{...} value}` line for
277    /// the given metric name) into a f64. Helpers only for tests.
278    fn parse_metric(text: &str, name: &str) -> f64 {
279        text.lines()
280            .find(|l| {
281                l.starts_with(name)
282                    && !l.starts_with(&format!("{name}_bucket"))
283                    && !l.starts_with(&format!("{name}_sum"))
284                    && !l.starts_with(&format!("{name}_count"))
285            })
286            .and_then(|l| l.split_whitespace().last())
287            .and_then(|v| v.parse::<f64>().ok())
288            .unwrap_or_else(|| panic!("metric {name} not found in:\n{text}"))
289    }

362    /// A service whose handler never resolves (client gone / proxy reset
363    /// mid-request). Dropping the request future must not leave the task in

376            Poll::Ready(Ok(()))

413    /// Regression: `sum + spent` must be monotonic even when a task is
414    /// cancelled via `TaskGuard::drop -> discard()`. Before the fix, `discard`
415    /// removed the task from `tasks` (the `sum` term) without crediting the
416    /// elapsed time to `stats_tasks_time_spend_total` (the `spent` term), so
417    /// `normalised` (and thus `rate()`) could drop, producing bogus spikes.
418    #[tokio::test]
419    async fn discarding_task_keeps_sum_plus_spent_monotonic() {
420        let kit = ServiceKit::new(MonitorServerConfig::default(), 4, Some(2));
421        let ready = Arc::new(AtomicBool::new(true));
422        kit.set_ready_probe(ready);
423
424        let metrics_after_start = {
425            let id = kit.task_tracker().start();
426            std::thread::sleep(std::time::Duration::from_millis(30));
427            // "Cancel" the task while it is still in flight: the guard would
428            // drop here, charging the elapsed time to `spent`.
429            let text = kit.metrics_text();
430            let sum_after_start = parse_metric(std::str::from_utf8(&text).unwrap(), "current_tasks_time_spend");
431            let spent_after_start = parse_metric(std::str::from_utf8(&text).unwrap(), "stats_tasks_time_spend_total");
432            (id, sum_after_start, spent_after_start)
433        };
434        let (id, sum_after_start, spent_after_start) = metrics_after_start;
435
436        // Simulate the guard's Drop -> discard().
437        std::thread::sleep(std::time::Duration::from_millis(30));
438        kit.task_tracker().discard(&id);
439
440        let text = kit.metrics_text();
441        let sum_after_discard = parse_metric(std::str::from_utf8(&text).unwrap(), "current_tasks_time_spend");
442        let spent_after_discard = parse_metric(std::str::from_utf8(&text).unwrap(), "stats_tasks_time_spend_total");
443
444        // The task left `tasks` (sum dropped) ...
445        assert!(
446            sum_after_discard < sum_after_start,
447            "sum should drop after discard: {sum_after_start} -> {sum_after_discard}"
448        );
449        // ... but its elapsed time was credited to `spent`, so the total
450        // (sum + spent) must not decrease.
451        assert!(
452            spent_after_discard > spent_after_start,
453            "spent should grow after discard: {spent_after_start} -> {spent_after_discard}"
454        );
455        assert!(
456            sum_after_discard + spent_after_discard >= sum_after_start + spent_after_start,
457            "sum+spent must be monotonic: {} + {} = {} vs {} + {} = {}",
458            sum_after_discard,
459            spent_after_discard,
460            sum_after_discard + spent_after_discard,
461            sum_after_start,
462            spent_after_start,
463            sum_after_start + spent_after_start,
464        );
465    }
