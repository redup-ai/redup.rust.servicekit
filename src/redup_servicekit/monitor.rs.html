1//! Monitoring core: task tracking, statistics and the main `ServiceKit`.
2//!
3//! Semantically identical to `redup.python.servicekit.monitoring`
4//! (`MonitorStorage` + `MonitorServer`): stats are accumulated in memory and
5//! the Prometheus registry is refreshed lazily on each `/metrics` request
6//! (`refresh()`), exactly like `refresh_registry_for_metrics()`.

17/// Distinct error type strings from `MonitorServer.errors` config (`service_errors`),
18/// used to classify a backend failure into `stats_errors{method,type}`. The string
19/// `"other"` is excluded because it is the default bucket, mirroring Python
20/// `ErrorParser._types`.
21fn collect_error_types(monitor: &MonitorServerConfig) -> Vec<String> {
22    monitor
23        .resolved_errors()
24        .values()
25        .flat_map(|e| e.service_errors.iter().cloned())
26        .filter(|t| !t.is_empty() && t != "other")
27        .collect()
28}

30/// Task outcome status (Python `TaskStatus`).
31#[derive(Debug, Clone, Copy, PartialEq, Eq)]
32pub enum TaskStatus {
33    Ok = 1,
34    Fatal = 2,
35    ConnectionError = 3,
36    OtherException = -1,
37    Internal = 4,
38    Unknown = 5,
39}
40
41impl TaskStatus {
42    pub fn name(&self) -> &'static str {
43        match self {
44            TaskStatus::Ok => "OK",
45            TaskStatus::Fatal => "FATAL",
46            TaskStatus::ConnectionError => "CONNECTION_ERROR",
47            TaskStatus::OtherException => "OTHER_EXCEPTION",
48            TaskStatus::Internal => "INTERNAL",
49            TaskStatus::Unknown => "UNKNOWN",
50        }
51    }
52}

54/// Internal statistics storage (analog of `MonitorStorage._stats`).
55#[derive(Debug, Default)]
56struct Stats {
57    /// Flat numeric stats (Python `_stats` counters), keyed by
58    /// `name___label__value...` strings.
59    counters: HashMap<String, f64>,
60    /// EMA long series (Python `stats_ema_long`).
61    ema_long: HashMap<String, f64>,
62    /// EMA short series (Python `stats_ema_short`).
63    ema_short: HashMap<String, f64>,
64    /// In-flight tasks: id -> start instant.
65    tasks: HashMap<String, Instant>,
66    /// Health status series: task_type -> (status name, series number).
67    statuses: HashMap<String, (String, u64)>,
68}
69
70/// Statistics store shared between task tracker and service kit.
71#[derive(Default)]
72struct StatsStore {
73    inner: Mutex<Stats>,
74    task_id_seq: AtomicU64,
75}
76
77impl StatsStore {
78    fn new() -> Self {
79        Self::default()
80    }
81
82    fn next_task_id(&self) -> String {
83        format!(
84            "task-{}-{}",
85            std::process::id(),
86            self.task_id_seq.fetch_add(1, Ordering::Relaxed)
87        )
88    }
89
90    fn inc(&self, key: &str, count: f64) {
91        let mut stats = self.inner.lock().unwrap();
92        *stats.counters.entry(key.to_string()).or_insert(0.0) += count;
93    }
94
95    fn set(&self, key: &str, value: f64) {
96        let mut stats = self.inner.lock().unwrap();
97        stats.counters.insert(key.to_string(), value);
98    }
99
100    /// Insert a stat unless the key already exists (idempotent init).
101    fn ensure(&self, key: &str, value: f64) {
102        let mut stats = self.inner.lock().unwrap();
103        stats.counters.entry(key.to_string()).or_insert(value);
104    }
105
106    fn observe(&self, key: &str, value: f64) {
107        let mut stats = self.inner.lock().unwrap();
108        // Split borrows: take maps out to avoid double mutable borrow.
109        let mut ema_long = std::mem::take(&mut stats.ema_long);
110        let mut ema_short = std::mem::take(&mut stats.ema_short);
111        crate::metrics::ema_update(&mut ema_long, &mut ema_short, key, value);
112        stats.ema_long = ema_long;
113        stats.ema_short = ema_short;
114    }
115
116    fn add_task(&self, id: String, start: Instant) {
117        let mut stats = self.inner.lock().unwrap();
118        stats.tasks.insert(id, start);
119    }
120
121    fn remove_task(&self, id: &str) -> Option<Duration> {
122        let mut stats = self.inner.lock().unwrap();
123        let start = stats.tasks.remove(id)?;
124        Some(start.elapsed())
125    }
126}

128/// Task tracker for recording in-flight tasks (analog of the `tasks` section
129/// in `MonitorStorage` plus `started_total`/`completed_total`/quantile metrics).
130#[derive(Clone)]
131pub struct TaskTracker {
132    store: Arc<StatsStore>,
133    registry: MetricsRegistry,
134}
135
136impl TaskTracker {
137    /// Begin a task: increments `started_total` and registers the in-flight task.
138    pub fn start(&self) -> String {
139        let id = self.store.next_task_id();
140        self.store.add_task(id.clone(), Instant::now());
141        self.registry
142            .inc_counter("started_total", "Total number of started tasks.", 1.0);
143        id
144    }
145
146    /// Finish a task: removes it, increments `completed_total`,
147    /// `stats_tasks_time_spend_total` and observes the quantile histogram.
148    pub fn finish(&self, id: &str) -> Option<Duration> {
149        let duration = self.store.remove_task(id)?;
150        self.registry
151            .inc_counter("completed_total", "Total number of completed tasks.", 1.0);
152        self.registry.inc_counter(
153            "stats_tasks_time_spend_total",
154            "Total time spent on completed tasks.",
155            duration.as_secs_f64(),
156        );
157        let hist = self.registry.histogram_vec(
158            "tasks_time_spent_quantile",
159            "Time spent on completing tasks.",
160            &[],
161            crate::metrics::BucketType::Time,
162        );
163        hist.with_label_values(&[self.registry.inittime()])
164            .observe(duration.as_secs_f64());
165        Some(duration)
166    }
167
168    /// Remove a task without metrics (e.g. on cancellation).
169    pub fn discard(&self, id: &str) {
170        self.store.remove_task(id);
171    }
172
173    /// Number of in-flight tasks.
174    pub fn in_flight(&self) -> usize {
175        self.store.inner.lock().unwrap().tasks.len()
176    }
177}

179/// Health readiness probe abstraction used by the HTTP `/readyz` endpoint.
180pub trait ReadyProbe: Send + Sync {
181    fn is_ready(&self) -> bool;
182}

184impl ReadyProbe for std::sync::atomic::AtomicBool {

186        self.load(Ordering::Relaxed)
187    }
188}

193    }

196/// Main entry point (analog of `MonitorServer`). Holds the metrics registry,
197/// statistics and config; refresh is called on `/metrics`.
198pub struct ServiceKit {
199    registry: MetricsRegistry,
200    monitor: MonitorServerConfig,
201    store: Arc<StatsStore>,
202    service_threads: u32,
203    task_tracker: TaskTracker,
204    ready_probe: Mutex<Option<Arc<dyn ReadyProbe>>>,
205    /// Distinct `service_errors` strings from config (minus `"other"`), for
206    /// classifying a backend failure into `stats_errors{method,type}`.
207    error_types: Vec<String>,
208}
209
210impl ServiceKit {
211    /// Create a `ServiceKit` (analog of `MonitorServer().run(...)`).
212    ///
213    /// `hpa_max_workers` is the target HPA pool size used to normalize the
214    /// task time stat; falls back to `max_workers` when `None`
215    /// (Python: `hpa_max_workers if hpa_max_workers is not None else max_workers`).
216    pub fn new(
217        monitor: MonitorServerConfig,
218        max_workers: u32,
219        hpa_max_workers: Option<u32>,
220    ) -> Arc<Self> {
221        let registry = MetricsRegistry::new(now_inittime());
222        let store = Arc::new(StatsStore::new());
223        let service_threads = hpa_max_workers.unwrap_or(max_workers);
224        let error_types = collect_error_types(&monitor);
225
226        let kit = Arc::new(Self {
227            registry: registry.clone(),
228            monitor,
229            store: store.clone(),
230            service_threads,
231            task_tracker: TaskTracker {
232                store: store.clone(),
233                registry: registry.clone(),
234            },
235            ready_probe: Mutex::new(None),
236            error_types,
237        });
238        kit.init_metrics();
239        kit
240    }

242    /// Initialize the fixed metric set (analog of `MonitorServer._init_metrics`).
243    fn init_metrics(&self) {
244        let r = &self.registry;
245        // Counters (names already include `_total` for wire compatibility).
246        r.counter_vec("started_total", "Total number of started tasks.", &[]);
247        r.counter_vec("completed_total", "Total number of completed tasks.", &[]);
248        r.counter_vec(
249            "stats_tasks_time_spend_total",
250            "Total time spent on completed tasks.",
251            &[],
252        );
253        // Gauges.
254        r.gauge_vec(
255            "tasks_count",
256            "The total number of tasks currently being processed.",
257            &[],
258        );
259        r.gauge_vec(
260            "current_tasks_time_spend",
261            "Total time spent processing current tasks.",
262            &[],
263        );
264        r.gauge_vec(
265            "current_tasks_max_time_spend",
266            "Maximum processing time for current tasks.",
267            &[],
268        );
269        r.gauge_vec(
270            "stats_tasks_with_current_time_spend_total",
271            "Total time spent processing current and previous tasks.",
272            &[],
273        );
274        r.gauge_vec(
275            "stats_tasks_with_current_time_spend_normalised_total",
276            "Total time spent on current and completed tasks per configured worker thread.",
277            &[],
278        );
279        r.gauge_vec("service_threads", "Number of service threads.", &[]);
280        // Histogram (existing in metrics contract; values appear on first finish).
281        r.histogram_vec(
282            "tasks_time_spent_quantile",
283            "Time spent on completing tasks.",
284            &[],
285            crate::metrics::BucketType::Time,
286        );
287        // Initial zero values (Python `.labels(...).inc(0)` / `.set(...)`).
288        let inittime = r.inittime();
289        r.gauge_vec("tasks_count", "", &[])
290            .with_label_values(&[inittime])
291            .set(0.0);
292        r.gauge_vec("current_tasks_time_spend", "", &[])
293            .with_label_values(&[inittime])
294            .set(0.0);
295        r.gauge_vec("current_tasks_max_time_spend", "", &[])
296            .with_label_values(&[inittime])
297            .set(0.0);
298        r.gauge_vec("stats_tasks_with_current_time_spend_total", "", &[])
299            .with_label_values(&[inittime])
300            .set(0.0);
301        r.gauge_vec(
302            "stats_tasks_with_current_time_spend_normalised_total",
303            "",
304            &[],
305        )
306        .with_label_values(&[inittime])
307        .set(0.0);
308        r.gauge_vec("service_threads", "", &[])
309            .with_label_values(&[inittime])
310            .set(self.service_threads as f64);
311        // Service info gauges (Python adds `service_threads` itself).
312        for (key, entry) in self
313            .monitor
314            .resolved_service_info(self.service_threads as f64)
315        {
316            let name = sanitize_metric_name(&key);
317            r.gauge_vec(&name, &entry.info, &[])
318                .with_label_values(&[inittime])
319                .set(entry.value);
320        }
321        // Pre-create per-method stats are done by services via `init_method`.
322    }

364    /// Pre-register per-method stats (analog of `grpc_init_wrapper`):
365    /// `request___method__M`, `processed___method__M___status__success|failure`,
366    /// `errors___method__M___type__<t>` for every configured error type.
367    pub fn init_method(&self, method: &str) {
368        self.ensure_stat(&format!("request___method__{method}"), 0.0);
369        self.ensure_stat(
370            &format!("processed___method__{method}___status__success"),
371            0.0,
372        );
373        self.ensure_stat(
374            &format!("processed___method__{method}___status__failure"),
375            0.0,
376        );
377        self.ensure_stat(&format!("errors___method__{method}___type__other"), 0.0);
378        for error_type in &self.error_types {
379            self.ensure_stat(
380                &format!("errors___method__{method}___type__{error_type}"),
381                0.0,
382            );
383        }
384    }

386    /// Classify a backend error message/text into an `stats_errors{type}` label.
387    ///
388    /// Mirrors Python `ErrorParser.parse`: any configured `service_errors`
389    /// string (case-insensitive substring) found in the text wins; otherwise
390    /// `"other"`. `"other"` itself is never a configured match, so it is the
391    /// fallback bucket.
392    pub fn classify_error(&self, text: &str) -> String {
393        let lower = text.to_lowercase();
394        if self.error_types.is_empty() {
395            return "other".to_string();
396        }
397        // Prefer the longlisted (first configured) match for determinism; exact
398        // substring scan over the configured set.
399        for error_type in &self.error_types {
400            if lower.contains(&error_type.to_lowercase()) {
401                return error_type.clone();
402            }
403        }
404        "other".to_string()
405    }

407    /// Observe a metric value: updates EMA long/short and, for keys with
408    /// `time`/`size` substrings, records a `stats_hist_*` observation.
409    pub fn observe_stat(&self, key: &str, value: f64) {
410        self.store.observe(key, value);
411        if let Some(bucket_type) = bucket_type_for_key(key) {
412            let (metric_name, labels) = parse_stat_key(key);
413            let hist_name = format!("stats_hist_{}", sanitize_metric_name(&metric_name));
414            let hist = self
415                .registry
416                .custom_histogram(&hist_name, &labels, bucket_type);
417            let values = custom_gauge_values(self.registry.inittime(), &labels);
418            let value_refs: Vec<&str> = values.iter().map(|s| s.as_str()).collect();
419            hist.with_label_values(&value_refs).observe(value);
420        }
421    }

428    /// Set a task status for health series (Python `set_task_status`).
429    pub fn set_task_status(&self, status: TaskStatus, task_type: &str) {
430        let mut stats = self.store.inner.lock().unwrap();
431        let (old_status, old_series) = stats
432            .statuses
433            .get(task_type)
434            .cloned()
435            .unwrap_or_else(|| (TaskStatus::Ok.name().to_string(), 0));
436        let (new_status, new_series) = if old_status == status.name() {
437            (old_status, old_series + 1)
438        } else {
439            (status.name().to_string(), 1)
440        };
441        stats
442            .statuses
443            .insert(task_type.to_string(), (new_status.clone(), new_series));
444        stats
445            .counters
446            .insert(format!("series.{task_type}"), new_series as f64);
447        tracing::debug!(task_type, status = %new_status, series = new_series, "task status updated");
448    }

450    /// Refresh the Prometheus registry from current stats (called on `/metrics`).
451    pub fn refresh(&self) {
452        let snapshot = {
453            let stats = self.store.inner.lock().unwrap();
454            (
455                stats.ema_long.clone(),
456                stats.ema_short.clone(),
457                stats.counters.clone(),
458                stats.tasks.clone(),
459            )
460        };
461        let (ema_long, ema_short, counters, tasks) = snapshot;
462        let now = Instant::now();
463
464        // EMA long/short -> stats_aggregation_<key> / stats_aggregation_last_<key>.
465        for (key, value) in &ema_long {
466            self.set_custom_stat(&format!("stats_aggregation_{key}"), *value);
467        }
468        for (key, value) in &ema_short {
469            self.set_custom_stat(&format!("stats_aggregation_last_{key}"), *value);
470        }
471        // Counters -> stats_<key> (gauge on the wire, as in Python).
472        for (key, value) in &counters {
473            if key.starts_with("series.") {
474                continue;
475            }
476            self.set_custom_stat(&format!("stats_{key}"), *value);
477        }
478        // Task series.
479        let task_durations: Vec<f64> = tasks
480            .values()
481            .map(|t| now.duration_since(*t).as_secs_f64())
482            .collect();
483        let tasks_count = task_durations.len() as f64;
484        let sum: f64 = task_durations.iter().sum();
485        let max: f64 = task_durations.iter().copied().fold(0.0, f64::max);
486        self.registry.set_gauge(
487            "tasks_count",
488            "The total number of tasks currently being processed.",
489            tasks_count,
490        );
491        self.registry.set_gauge(
492            "current_tasks_time_spend",
493            "Total time spent processing current tasks.",
494            sum,
495        );
496        self.registry.set_gauge(
497            "current_tasks_max_time_spend",
498            "Maximum processing time for current tasks.",
499            max,
500        );
501
502        // Combined spent + current (monotonic gauge), then normalized.
503        let spent = self
504            .registry
505            .counter_value("stats_tasks_time_spend_total")
506            .unwrap_or(0.0);
507        let combined = sum + spent;
508        self.registry.set_gauge(
509            "stats_tasks_with_current_time_spend_total",
510            "Total time spent processing current and previous tasks.",
511            combined,
512        );
513        let normalised = if self.service_threads > 0 {
514            combined / self.service_threads as f64
515        } else {
516            0.0
517        };
518        self.registry.set_gauge(
519            "stats_tasks_with_current_time_spend_normalised_total",
520            "Total time spent on current and completed tasks per configured worker thread.",
521            normalised,
522        );
523        self.registry.set_gauge(
524            "service_threads",
525            "Number of service threads.",
526            self.service_threads as f64,
527        );
528    }
529
530    fn set_custom_stat(&self, full_key: &str, value: f64) {
531        // full_key may be "stats_request___method__Foo" or
532        // "stats_aggregation_time full___method__Foo".
533        let (metric_name, labels) = parse_stat_key(full_key);
534        let sanitized = sanitize_metric_name(&metric_name);
535        let vec = self.registry.custom_gauge(&sanitized, &labels);
536        let values = custom_gauge_values(self.registry.inittime(), &labels);
537        let value_refs: Vec<&str> = values.iter().map(|s| s.as_str()).collect();
538        vec.with_label_values(&value_refs).set(value);
539    }

541    /// Export metrics in Prometheus text format (with a preceding refresh).
542    pub fn metrics_text(&self) -> Vec<u8> {
543        self.refresh();

545    }

547    /// Current statuses (health series), analog of `get_statuses`.
